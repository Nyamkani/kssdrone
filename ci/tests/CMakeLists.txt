cmake_minimum_required(VERSION 3.21)
project(kssdrone_host_tests LANGUAGES CXX)
enable_testing()
option(ENABLE_COVERAGE "Enable GNU coverage" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
function(add_drone_test name source label)
    add_executable(${name} ${source})
    target_include_directories(${name} PRIVATE ../../main support)
    target_compile_options(${name} PRIVATE -Wall -Wextra -UNDEBUG)
    if(ENABLE_COVERAGE)
        target_compile_options(${name} PRIVATE -O0 -g --coverage)
        target_link_options(${name} PRIVATE --coverage)
    endif()
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES LABELS ${label} TIMEOUT 15)
endfunction()
add_drone_test(unit_pid unit/pid.cpp unit)
add_drone_test(unit_mixer unit/mixer.cpp unit)
add_drone_test(unit_estimator unit/estimator.cpp unit)
add_drone_test(integration_control integration/control.cpp integration)
add_drone_test(regression_control regression/control.cpp regression)
add_drone_test(sil_rate simulation/rate.cpp simulation)
